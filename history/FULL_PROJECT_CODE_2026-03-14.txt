void drawGlyphScaled(int x, int y, char c,
                     const BDFont &font,
                     uint16_t color,
                     uint8_t scale)
{
    if (c < font.charFirst || c > font.charLast) return;
    const BDGlyph &g = font.glyphs[c - font.charFirst];

    for (uint8_t row = 0; row < g.height; row++) {
        uint8_t bits = g.bitmap[row];
        for (uint8_t col = 0; col < g.width; col++) {
            if (bits & (0x80 >> col)) {
                for (uint8_t dy = 0; dy < scale; dy++)
                for (uint8_t dx = 0; dx < scale; dx++)
                    drawPixel_FAST(x + col*scale + dx, y + row*scale + dy, color);
            }
        }
    }
}

// =======================================================
// drawTextScaled â€” final fixed ordering
// =======================================================

void drawTextScaled(int x, int y, const char *text,
                    const BDFont &font,
                    uint16_t color,
                    uint8_t scale)
{
    if (!text || scale == 0) return;
    int cx = x;

    while (*text) {
        unsigned char c = *text;

        // ---- 1) Percent â€” highest priority (must be first) ----
        if (c == '%') {
            const BDGlyph &g = glyph_percent;
            for (uint8_t row = 0; row < g.height; row++) {
                uint8_t bits = g.bitmap[row];
                for (uint8_t col = 0; col < g.width; col++) {
                    if (bits & (0x80 >> col)) {
                        for (uint8_t dy = 0; dy < scale; dy++)
                        for (uint8_t dx = 0; dx < scale; dx++)
                            drawPixel_FAST(cx + col*scale + dx, y + row*scale + dy, color);
                    }
                }
            }
            cx += (glyph_percent.width + 1) * scale;
            text++;
            continue;
        }

        // ---- 2) UTF-8 degree symbol 0xC2 0xB0 ----
        if (c == 0xC2 && (unsigned char)text[1] == 0xB0) {
            const BDGlyph &g = glyph_degree;
            for (uint8_t row = 0; row < g.height; row++) {
                uint8_t bits = g.bitmap[row];
                for (uint8_t col = 0; col < g.width; col++) {
                    if (bits & (0x80 >> col)) {
                        for (uint8_t dy = 0; dy < scale; dy++)
                        for (uint8_t dx = 0; dx < scale; dx++)
                            drawPixel_FAST(cx + col*scale + dx, y + row*scale + dy, color);
                    }
                }
            }
            cx += (g.width + 1) * scale;
            text += 2;
            continue;
        }

        // ---- 3) Single-byte ANSI degree (rare but supported) ----
        if (c == 0xB0 || c == 0xF8) {
            const BDGlyph &g = glyph_degree;
            for (uint8_t row = 0; row < g.height; row++) {
                uint8_t bits = g.bitmap[row];
                for (uint8_t col = 0; col < g.width; col++) {
                    if (bits & (0x80 >> col)) {
                        for (uint8_t dy = 0; dy < scale; dy++)
                        for (uint8_t dx = 0; dx < scale; dx++)
                            drawPixel_FAST(cx + col*scale + dx, y + row*scale + dy, color);
                    }
                }
            }
            cx += (glyph_degree.width + 1) * scale;
            text++;
            continue;
        }

        // ---- 4) Slash ----
        if (c == '/') {
            const BDGlyph &g = glyph_slash;
            for (uint8_t row = 0; row < g.height; row++) {
                uint8_t bits = g.bitmap[row];
                for (uint8_t col = 0; col < g.width; col++) {
                    if (bits & (0x80 >> col)) {
                        for (uint8_t dy = 0; dy < scale; dy++)
                        for (uint8_t dx = 0; dx < scale; dx++)
                            drawPixel_FAST(cx + col*scale + dx, y + row*scale + dy, color);
                    }
                }
            }
            cx += (glyph_slash.width + 1) * scale;
            text++;
            continue;
        }

        // ---- 5) Space ----
        if (c == ' ') {
            cx += (font.glyphs[0].width + 1) * scale;
            text++;
            continue;
        }

        // ---- 6) Normal ASCII glyph ----
        if (c >= font.charFirst && c <= font.charLast) {
            const BDGlyph &g = font.glyphs[c - font.charFirst];
            drawGlyphScaled(cx, y, c, font, color, scale);
            cx += (g.width + 1) * scale;
            text++;
            continue;
        }

        // ---- 7) Fallback skip ----
        cx += 2 * scale;
        text++;
    }
}

// =======================================================
// measureTextWidth â€” supports all symbols
// =======================================================

int measureTextWidth(const char *text,
                     const BDFont &font,
                     uint8_t scale)
{
    if (!text) return 0;
    int px = 0;

    while (*text) {
        unsigned char c = *text++;

        if (c == '%')      { px += (glyph_percent.width + 1) * scale; continue; }
        if (c == 0xC2 && (unsigned char)*text == 0xB0) { px += (glyph_degree.width + 1) * scale; text++; continue; }
        if (c == 0xB0 || c == 0xF8) { px += (glyph_degree.width + 1) * scale; continue; }
        if (c == '/')      { px += (glyph_slash.width + 1) * scale; continue; }
        if (c == ' ')      { px += (font.glyphs[0].width + 1) * scale; continue; }

        if (c >= font.charFirst && c <= font.charLast) {
            const BDGlyph &g = font.glyphs[c - font.charFirst];
            px += (g.width + 1) * scale;
        } else {
            px += 2 * scale;
        }
    }

    return px;
}

// =======================================================
// Centers VALUE + UNIT block
// =======================================================

void drawValueUnitCentered(int x0, int y0, int x1, int y1,
                           const char *value,
                           uint8_t fieldDigits,
                           const char *unit,
                           const BDFont &fontValue,
                           uint8_t scaleValue,
                           const BDFont &fontUnit,
                           uint8_t scaleUnit,
                           uint16_t color)
{
    int digitW = (fontValue.glyphs[0].width + 1) * scaleValue;
    int valueW = digitW * strlen(value);
    int unitW  = measureTextWidth(unit, fontUnit, scaleUnit);

    int totalWidth = valueW + (unitW > 0 ? (digitW / 2) : 0) + unitW;
    int cx = (x0 + x1) / 2;
    int left = cx - totalWidth / 2;

    int valueTop = y0 + 20;
    drawTextScaled(left, valueTop, value, fontValue, color, scaleValue);

    if (unitW > 0) {
        int unitTop = valueTop + 7 * scaleValue - 7 * scaleUnit;
        drawTextScaled(left + valueW + (digitW / 2), unitTop, unit, fontUnit, color, scaleUnit);
    }
}

#include "Display_ST7796S.h"

// ===== Private SPI write helpers =====
static void wr_cmd(uint8_t c) {
    digitalWrite(TFT_DC, LOW);
    digitalWrite(TFT_CS, LOW);
    SPI.transfer(c);
    digitalWrite(TFT_CS, HIGH);
}

static void wr_dat(uint8_t d) {
    digitalWrite(TFT_DC, HIGH);
    digitalWrite(TFT_CS, LOW);
    SPI.transfer(d);
    digitalWrite(TFT_CS, HIGH);
}

static void wr_dat16(uint16_t d) {
    wr_dat(d >> 8);
    wr_dat(d & 0xFF);
}

// ===== Display init (known-good from loader) =====
void initDisplay() {
    pinMode(TFT_CS,  OUTPUT);
    pinMode(TFT_DC,  OUTPUT);
    pinMode(TFT_RST, OUTPUT);
    pinMode(TFT_LED, OUTPUT);
    digitalWrite(TFT_LED, HIGH);

    SPI.begin();
    SPI.beginTransaction(SPISettings(16000000, MSBFIRST, SPI_MODE3));

    // Hardware reset
    digitalWrite(TFT_RST, LOW);
    delay(40);
    digitalWrite(TFT_RST, HIGH);
    delay(120);

    // ST7796S init
    wr_cmd(0xF0); wr_dat(0xC3);
    wr_cmd(0xF0); wr_dat(0x96);
    wr_cmd(0x36); wr_dat(0x28);   // landscape
    wr_cmd(0x3A); wr_dat(0x55);   // RGB565
    wr_cmd(0xB4); wr_dat(0x01);
    wr_cmd(0xB7); wr_dat(0xC6);
    wr_cmd(0x11);
    delay(120);
    wr_cmd(0x29);

    // Default drawing window full screen
    wr_cmd(0x2A); wr_dat16(0); wr_dat16(479);
    wr_cmd(0x2B); wr_dat16(0); wr_dat16(319);
    wr_cmd(0x2C);
}

// ===== Set drawing window =====
void setAddrWindow(uint16_t x0, uint16_t y0, uint16_t x1, uint16_t y1) {
    wr_cmd(0x2A);
    wr_dat16(x0);
    wr_dat16(x1);

    wr_cmd(0x2B);
    wr_dat16(y0);
    wr_dat16(y1);

    wr_cmd(0x2C);
}

// ===== Draw single pixel =====
void drawPixel_FAST(int x, int y, uint16_t color) {
    if (x < 0 || x >= 480 || y < 0 || y >= 320) return;

    setAddrWindow(x, y, x, y);
    wr_dat16(color);
}

// ===== FULL, INSTANT SCREEN FILL (NO WIPE) =====
void fillScreen_FAST(uint16_t color) {
    // full window once
    setAddrWindow(0, 0, 479, 319);

    uint8_t hi = color >> 8;
    uint8_t lo = color & 0xFF;

    digitalWrite(TFT_DC, HIGH);
    digitalWrite(TFT_CS, LOW);

    // blast the entire framebuffer with one continuous stream
    uint32_t total = 480UL * 320UL;
    while (total--) {
        SPI.transfer(hi);
        SPI.transfer(lo);
    }

    digitalWrite(TFT_CS, HIGH);
}

// ===== LED Backlight PWM (0â€“255) =====
void setBacklightPWM(uint8_t value) {
    analogWrite(TFT_LED, value);
}

// ---------------------------------------------------------------------------
// FULL ICON RAW DRAW FUNCTION (embedded by request â€” independent of RawImage.h)
// Draws a 96Ã—72 RGB565 little-endian icon at (x, y)
// ---------------------------------------------------------------------------
static void drawRAW_icon96x72(const char* filename, int x, int y)
{
    File f = SD.open(filename, FILE_READ);
    if (!f) return;

    const int BUFSIZE = 512;
    uint8_t buf[BUFSIZE];
    int n = f.read(buf, BUFSIZE);
    int px = 0;

    for (uint16_t yy = 0; yy < ICON_H; yy++)
    {
        for (uint16_t xx = 0; xx < ICON_W; xx++)
        {
            if (px + 1 >= n) {
                n = f.read(buf, BUFSIZE);
                px = 0;
                if (!n) { f.close(); return; }
            }

            // RGB565 little-endian: low byte first, then high byte
            uint8_t lo = buf[px++];
            uint8_t hi = buf[px++];
            uint16_t color = (hi << 8) | lo;

            drawPixel_FAST(x + xx, y + yy, color);
        }
    }

    f.close();
}

    // Reserved for future features (blink timing, animation, etc.)

    int x = 0;
    for (int i = 0; i < 5; i++) {
        drawRAW_icon96x72(ICON_FILES[i], x, ICON_Y);
        x += ICON_W;
    }
}

    pageUpdate[PAGE_BOOTLOGO]  = Page_BootLogo_update;

    pageBegin[PAGE_ENV]        = Page_Env_begin;
    pageUpdate[PAGE_ENV]       = Page_Env_update;

    pageBegin[PAGE_PID1]       = Page_PID1_begin;
    pageUpdate[PAGE_PID1]      = Page_PID1_update;

    pageBegin[PAGE_PID2]       = Page_PID2_begin;
    pageUpdate[PAGE_PID2]      = Page_PID2_update;

    pageBegin[PAGE_PID3]       = Page_PID3_begin;
    pageUpdate[PAGE_PID3]      = Page_PID3_update;

    pageBegin[PAGE_DTC]        = Page_DTC_begin;
    pageUpdate[PAGE_DTC]       = Page_DTC_update;

    currentPage = PAGE_BOOTLOGO;
    lastPageChangeTime = millis();
    if (pageBegin[currentPage]) pageBegin[currentPage]();
    Touch_reset();   // <<< ensures first page also starts with no "finger down"
    blockingInputUntil = millis() + 200; // Block touch input briefly to avoid "switch during touch" ghost box

}

void PageManager_update()
{
    // Ignore all touch input during the post-switch guard window
    if (millis() < blockingInputUntil) {
        Touch_reset();
        return;
    }

    if (pageChangePending) {
        pageChangePending = false;
        currentPage = nextPage;
        lastPageChangeTime = millis();
        if (pageBegin[currentPage]) {
            pageBegin[currentPage]();
        }
        Touch_reset();
        blockingInputUntil = millis() + 200;
    }

    if (millis() - lastPageChangeTime >= PAGE_CHANGE_DEBOUNCE_MS) {
        if (pageUpdate[currentPage]) {
            pageUpdate[currentPage]();
        }
    }
}

void PageManager_switchTo(PageID id)
{
    if (id >= PAGE_MAX) return;
    nextPage = id;
    pageChangePending = true;
}

PageID PageManager_current()
{
    return currentPage;
}

#include "Page_BootLogo.h"
#include "Display_ST7796S.h"
#include "RawImage.h"
#include "Touch_XPT2046.h"
#include "PageManager.h"

static uint32_t bootStartTime = 0;
static bool fadeActive = false;
static uint8_t fadeStep = 0;

#define BOOT_TIMEOUT_MS   2500       // 2.5 seconds before auto-exit
#define FADE_STEPS        32
#define FADE_INTERVAL_MS  25         // 32 Ã— 25ms â‰ˆ 800ms fade-out

void Page_BootLogo_begin()
{
    bootStartTime = millis();
    fadeActive = false;
    fadeStep = 0;

    setBacklightPWM(255);               // full brightness
    fillScreen_FAST(0x0000);
    drawRAW480x320("/bowtie.raw");
}

void Page_BootLogo_update()
{
    uint32_t now = millis();

    // Trigger fade from either touch or timeout
    TouchPoint tp = readTouch();
    if (!fadeActive && (tp.touched || (now - bootStartTime >= BOOT_TIMEOUT_MS)))
    {
        fadeActive = true;
        fadeStep = 0;
    }

    // Run fade non-blocking
    if (fadeActive)
    {
        static uint32_t lastFade = 0;
        if (now - lastFade >= FADE_INTERVAL_MS)
        {
            lastFade = now;
            fadeStep++;

            // linear brightness: 255 â†’ 0
            int brightness = 255 - ((255 * fadeStep) / FADE_STEPS);
            if (brightness < 0) brightness = 0;
            setBacklightPWM(brightness);

            // fade finished
            if (fadeStep >= FADE_STEPS)
            {
                setBacklightPWM(0);                // keep screen dark
                fadeActive = false;
setBacklightPWM(0);            // stay dark until ENV fade-in
PageManager_switchTo(PAGE_ENV);
                return;
            }
        }
    }
}

#include "RawImage.h"
#include <SD.h>
#include "Display_ST7796S.h"  // for drawPixel_FAST(), etc.

// Constants for the RAW boot image
static const uint16_t RAW_WIDTH  = 480;
static const uint16_t RAW_HEIGHT = 320;

// Size of the streaming buffer (must be multiple of 2 for RGB565)
static const size_t RAW_BUFFER_SIZE = 512;

void drawRAW480x320(const char *filename)
{
    Serial.print("Opening raw: ");
    Serial.println(filename);

    // Try to open the file from SD
    File f = SD.open(filename, FILE_READ);
    if (!f) {
        // If it fails, just bail out quietly (screen stays whatever it was)
        // You could also fillScreen_FAST with a solid color here if you want.
        return;
    }

    uint8_t  buffer[RAW_BUFFER_SIZE];
    size_t   bytesInBuffer = 0;
    size_t   bufferIndex   = 0;

    // Helper lambda to refill the buffer when needed
    auto refillBuffer = [&]() -> bool {
        bytesInBuffer = f.read(buffer, RAW_BUFFER_SIZE);
        bufferIndex   = 0;
        return (bytesInBuffer > 0);
    };

    // Prime the buffer
    if (!refillBuffer()) {
        f.close();
        return;
    }

    // Stream through the file and draw pixels row by row
    for (uint16_t y = 0; y < RAW_HEIGHT; y++) {
        for (uint16_t x = 0; x < RAW_WIDTH; x++) {

            // Ensure we have at least 2 bytes available for one pixel
            if (bufferIndex + 1 >= bytesInBuffer) {
                if (!refillBuffer()) {
                    // File ended unexpectedly; stop drawing
                    f.close();
                    return;
                }
            }

            uint8_t hi = buffer[bufferIndex++];
            uint8_t lo = buffer[bufferIndex++];

            // RAW is little-endian RGB565: hi byte first, then lo byte
            uint16_t color = (uint16_t)(hi << 8) | lo;

            // Draw pixel at (x, y) in the current display orientation
            drawPixel_FAST(x, y, color);
        }
    }

    f.close();
}

#include "Touch_XPT2046.h"
#include "PinConfig.h"
#include <SPI.h>

// Raw XPT2046 instance
static XPT2046_Touchscreen ts(TOUCH_CS, TOUCH_IRQ);

// Internal edge detection state
static bool lastPressed = false;

// Reset touch state (called by PageManager after a page switch)
void Touch_reset() {
    lastPressed = false;
}

void initTouch() {
    ts.begin();
    ts.setRotation(3);   // match display orientation
    Touch_reset();       // ensure clean start
}

// This returns ONE press event per real tap (edge-triggered)
TouchPoint readTouch() {
    bool pressed = ts.touched();

    // Rising edge = NEW PRESS â†’ generate touch event ONCE
    if (pressed && !lastPressed) {
        lastPressed = true;

        TS_Point p = ts.getPoint();

        const int RAW_MIN = 200;
        const int RAW_MAX = 3800;

        int16_t sx = map(p.x, RAW_MIN, RAW_MAX, 0, 479);
        int16_t sy = map(p.y, RAW_MIN, RAW_MAX, 0, 319);

        if (sx < 0)   sx = 0;
        if (sx > 479) sx = 479;
        if (sy < 0)   sy = 0;
        if (sy > 319) sy = 319;

        return { sx, sy, true };   // â† ONE event per tap
    }

    // Falling edge â€” release â†’ reset state ONLY
    if (!pressed && lastPressed) {
        lastPressed = false;
    }

    // Anything else â†’ no event
    return { 0, 0, false };
}

bool touchPressed() {
    // Page updates should only care whether a fresh press event is available
    TouchPoint tp = readTouch();
    return tp.touched;
}
